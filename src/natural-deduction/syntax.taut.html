260// ctx is the precedence the surroundings require; a weaker formula gets parentheses.
261fn render_at(a: Fm, ctx: Nat): Text {
262  let need := Nat::lt(prec(a), ctx)
263  match a {
264    Atom(n) => Text::concat("P", Text::of_nat(n))
265    Bot => "⊥"
266    Not(b) => Text::concat("¬", render_at(b, 3))
267    And(b, c) => paren(Text::concat(render_at(b, 2), Text::concat(" ∧ ", render_at(c, 3))), need)
268    Or(b, c) => paren(Text::concat(render_at(b, 1), Text::concat(" ∨ ", render_at(c, 2))), need)
269    Imp(b, c) => paren(Text::concat(render_at(b, 1), Text::concat(" → ", render_at(c, 0))), need)
270  }

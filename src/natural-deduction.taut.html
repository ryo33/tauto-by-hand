3pub(open) inductive Mem: (Fm, Ctx()) -> Prop {
4  here<a: Fm, Γ: Ctx()>: Mem(a, [a, ..Γ])
5  there<a: Fm, b: Fm, Γ: Ctx()>(h: Mem(a, Γ)): Mem(a, [b, ..Γ])
6}
7
8pub(open) inductive Derivable: (Ctx(), Fm) -> Prop {
9  ax<Γ: Ctx(), a: Fm>(h: Mem(a, Γ)): f!`Γ ⊢ a`
10  and_intro<Γ: Ctx(), a: Fm, b: Fm>(ha: f!`Γ ⊢ a`, hb: f!`Γ ⊢ b`): f!`Γ ⊢ a ∧ b`
11  and_elim_left<Γ: Ctx(), a: Fm, b: Fm>(h: f!`Γ ⊢ a ∧ b`): f!`Γ ⊢ a`
12  and_elim_right<Γ: Ctx(), a: Fm, b: Fm>(h: f!`Γ ⊢ a ∧ b`): f!`Γ ⊢ b`
13  imp_intro<Γ: Ctx(), a: Fm, b: Fm>(h: f!`a, Γ ⊢ b`): f!`Γ ⊢ a → b`
14  imp_elim<Γ: Ctx(), a: Fm, b: Fm>(hf: f!`Γ ⊢ a → b`, ha: f!`Γ ⊢ a`): f!`Γ ⊢ b`
15  or_intro_left<Γ: Ctx(), a: Fm, b: Fm>(h: f!`Γ ⊢ a`): f!`Γ ⊢ a ∨ b`
16  or_intro_right<Γ: Ctx(), a: Fm, b: Fm>(h: f!`Γ ⊢ b`): f!`Γ ⊢ a ∨ b`
17  or_elim<Γ: Ctx(), a: Fm, b: Fm, c: Fm>(
18    h: f!`Γ ⊢ a ∨ b`,
19    ha: f!`a, Γ ⊢ c`,
20    hb: f!`b, Γ ⊢ c`
21  ): f!`Γ ⊢ c`
22  not_intro<Γ: Ctx(), a: Fm>(h: f!`a, Γ ⊢ ⊥`): f!`Γ ⊢ ¬a`
23  not_elim<Γ: Ctx(), a: Fm>(hn: f!`Γ ⊢ ¬a`, ha: f!`Γ ⊢ a`): f!`Γ ⊢ ⊥`
24  bot_elim<Γ: Ctx(), a: Fm>(h: f!`Γ ⊢ ⊥`): f!`Γ ⊢ a`
25}
26
27theorem imp_self(a: Fm): f!`⊢ a → a` {
28  let h: f!`a ⊢ a` := ax(here)
29  imp_intro(h)
30}
31
32theorem imp_const(a: Fm, b: Fm): f!`⊢ a → b → a` {
33  let h1: f!`b, a ⊢ a` := ax(there(here))
34  let h2: f!`a ⊢ b → a` := imp_intro(h1)
35  imp_intro(h2)
36}
37
38theorem and_swap(a: Fm, b: Fm): f!`⊢ a ∧ b → b ∧ a` {
39  let h: f!`a ∧ b ⊢ a ∧ b` := ax(here)
40  let hb: f!`a ∧ b ⊢ b` := and_elim_right(h)
41  let ha: f!`a ∧ b ⊢ a` := and_elim_left(h)
42  imp_intro(and_intro(hb, ha))
43}
44
45theorem or_swap(a: Fm, b: Fm): f!`⊢ a ∨ b → b ∨ a` {
46  let h: f!`a ∨ b ⊢ a ∨ b` := ax(here)
47  let ha: f!`a, a ∨ b ⊢ b ∨ a` := or_intro_right(ax(here))
48  let hb: f!`b, a ∨ b ⊢ b ∨ a` := or_intro_left(ax(here))
49  imp_intro(or_elim(h, ha, hb))
50}
51
52theorem dni(a: Fm): f!`⊢ a → ¬¬a` {
53  let hn: f!`¬a, a ⊢ ¬a` := ax(here)
54  let ha: f!`¬a, a ⊢ a` := ax(there(here))
55  let hbot: f!`¬a, a ⊢ ⊥` := not_elim(hn, ha)
56  let hdn: f!`a ⊢ ¬¬a` := not_intro(hbot)
57  imp_intro(hdn)
58}

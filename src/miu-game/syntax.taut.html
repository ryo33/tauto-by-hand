6inductive_syntax!(miu, ```par

10Word: { Piece };

14ILetter: 'I';
15ULetter: 'U';

23pub fn 連結(a: List(記号), b: List(記号)): List(記号) {
24  match a {
25    [] => b
26    [x, ..rest] => [x, ..連結(rest, b)]
27  }
28}

30fn 列(ps: List(miu::Piece)): SynExpr {
31  match ps {
32    [] => quote { [] }
33    [p, ..rest] => {
34      let r := 列(rest)
35      match p {
36        Letter(l) => match l {
37          MLetter(_) => quote { [M, ..$r] }
38          ILetter(_) => quote { [I, ..$r] }
39          ULetter(_) => quote { [U, ..$r] }
40        }
41        Var(t) => {
42          let x := ExVar(Syn::ident(t))
43          match rest {
44            [] => x
45            _ => quote { 連結($x, $r) }
46          }
47        }
48      }
49    }
50  }
51}

53pub macro fn m(w: miu::Word): SynExpr {
54  列(w.piece)
